     1	# ============================================================
     2	# A15B RANKED INTERACTIVE eEF1A NETWORK
     3	#
     4	# Uses the provenance-corrected A13B network and A15 rankings.
     5	# Adds rank-sensitive compound styling, tooltips, two ranking
     6	# tables below the graph, and a dedicated ranking-visual key.
     7	#
     8	# Output:
     9	#   results/eef1a_network_A15_ranked.html
    10	# ============================================================
    11	
    12	required_packages <- c("readr", "dplyr", "visNetwork", "htmlwidgets", "htmltools")
    13	missing_packages <- required_packages[
    14	  !vapply(required_packages, requireNamespace, logical(1), quietly = TRUE)
    15	]
    16	if (length(missing_packages) > 0) stop(paste("Missing packages:", paste(missing_packages, collapse = ", ")))
    17	
    18	results_dir <- file.path(normalizePath(getwd(), winslash = "/", mustWork = TRUE), "results")
    19	files <- list(
    20	  proteins = file.path(results_dir, "nodes_proteins.csv"),
    21	  pathways = file.path(results_dir, "nodes_pathways.csv"),
    22	  go = file.path(results_dir, "nodes_biological_process.csv"),
    23	  compounds = file.path(results_dir, "nodes_drugs.csv"),
    24	  ppi = file.path(results_dir, "edges_interacts_with.csv"),
    25	  protein_pathway = file.path(results_dir, "edges_protein_pathway.csv"),
    26	  protein_go = file.path(results_dir, "edges_protein_go.csv"),
    27	  compound_edges = file.path(results_dir, "edges_compound_protein_corrected.csv"),
    28	  scores = file.path(results_dir, "A15_candidate_score_components.csv"),
    29	  direct = file.path(results_dir, "A15_direct_eef1a1_ranking.csv"),
    30	  network = file.path(results_dir, "A15_translation_network_ranking.csv")
    31	)
    32	output_file <- file.path(results_dir, "eef1a_network_A15_ranked.html")
    33	missing_files <- unlist(files)[!file.exists(unlist(files))]
    34	if (length(missing_files) > 0) stop(paste0("Missing inputs:\n", paste(missing_files, collapse = "\n")))
    35	
    36	read_data <- function(path) readr::read_csv(path, show_col_types = FALSE, progress = FALSE)
    37	proteins <- read_data(files$proteins)
    38	pathways <- read_data(files$pathways)
    39	go_processes <- read_data(files$go)
    40	compounds <- read_data(files$compounds)
    41	ppi <- read_data(files$ppi)
    42	protein_pathway <- read_data(files$protein_pathway)
    43	protein_go <- read_data(files$protein_go)
    44	compound_edges_data <- read_data(files$compound_edges)
    45	scores <- read_data(files$scores)
    46	direct_ranking <- read_data(files$direct)
    47	network_ranking <- read_data(files$network)
    48	
    49	# Merge ranks and scores into compound nodes.
    50	compound_rank_data <- compounds |>
    51	  dplyr::left_join(scores, by = "molecule_chembl_id") |>
    52	  dplyr::left_join(
    53	    direct_ranking |> dplyr::select(molecule_chembl_id, direct_rank),
    54	    by = "molecule_chembl_id"
    55	  ) |>
    56	  dplyr::left_join(
    57	    network_ranking |> dplyr::select(molecule_chembl_id, network_rank),
    58	    by = "molecule_chembl_id"
    59	  ) |>
    60	  dplyr::mutate(
    61	    name = dplyr::coalesce(name, molecule_chembl_id),
    62	    overall_priority_score = dplyr::coalesce(overall_priority_score, 0),
    63	    compound_size = 18 + 1.6 * overall_priority_score,
    64	    is_top_direct = direct_rank == min(direct_rank, na.rm = TRUE),
    65	    is_top_network = network_rank == min(network_rank, na.rm = TRUE),
    66	    compound_border_width = dplyr::case_when(
    67	      is_top_direct & is_top_network ~ 7,
    68	      is_top_direct ~ 6,
    69	      is_top_network ~ 4,
    70	      TRUE ~ 2
    71	    ),
    72	    compound_shadow_size = dplyr::case_when(
    73	      is_top_network ~ 18,
    74	      TRUE ~ 5
    75	    ),
    76	    compound_shadow_color = dplyr::case_when(
    77	      is_top_network ~ "rgba(70,35,120,0.55)",
    78	      TRUE ~ "rgba(0,0,0,0.16)"
    79	    ),
    80	    rank_label = paste0("D", direct_rank, " / N", network_rank, "  ", name)
    81	  )
    82	
    83	# Connectivity calculations for biological nodes.
    84	ppi <- ppi |> dplyr::mutate(score = dplyr::coalesce(suppressWarnings(as.numeric(score)), 0))
    85	ppi_degree <- table(c(ppi$source, ppi$target))
    86	ppi_degree <- data.frame(protein = names(ppi_degree), ppi_degree = as.numeric(ppi_degree))
    87	pathway_degree <- protein_pathway |> dplyr::count(protein, name = "pathway_degree")
    88	go_degree <- protein_go |> dplyr::count(protein, name = "go_degree")
    89	compound_degree <- compound_edges_data |> dplyr::count(queried_protein, name = "compound_degree") |> dplyr::rename(protein = queried_protein)
    90	
    91	protein_metrics <- proteins |>
    92	  dplyr::left_join(ppi_degree, by = "protein") |>
    93	  dplyr::left_join(pathway_degree, by = "protein") |>
    94	  dplyr::left_join(go_degree, by = "protein") |>
    95	  dplyr::left_join(compound_degree, by = "protein") |>
    96	  dplyr::mutate(
    97	    dplyr::across(c(ppi_degree, pathway_degree, go_degree, compound_degree), ~ dplyr::coalesce(.x, 0)),
    98	    node_size = 18 + 3 * sqrt(ppi_degree + pathway_degree + go_degree + compound_degree)
    99	  )
   100	
   101	pathways <- pathways |>
   102	  dplyr::left_join(protein_pathway |> dplyr::count(pathway, name = "protein_count"), by = "pathway") |>
   103	  dplyr::mutate(protein_count = dplyr::coalesce(protein_count, 0L), node_size = 10 + 2.3 * sqrt(protein_count))
   104	
   105	go_processes <- go_processes |>
   106	  dplyr::left_join(protein_go |> dplyr::count(biological_process, name = "protein_count"), by = c("go_term" = "biological_process")) |>
   107	  dplyr::mutate(protein_count = dplyr::coalesce(protein_count, 0L), node_size = 9 + 2.1 * sqrt(protein_count))
   108	
   109	protein_nodes <- protein_metrics |>
   110	  dplyr::transmute(
   111	    id = paste0("protein::", protein), label = protein, group = "Protein", shape = "dot", size = node_size,
   112	    title = paste0("<b>Protein:</b> ", protein, "<br><b>Compound links:</b> ", compound_degree)
   113	  )
   114	pathway_nodes <- pathways |>
   115	  dplyr::transmute(
   116	    id = paste0("pathway::", pathway), label = pathway, group = "Pathway", shape = "dot", size = node_size,
   117	    title = paste0("<b>Reactome pathway:</b> ", pathway, "<br><b>Adjusted p:</b> ", signif(p_adjust, 3))
   118	  )
   119	go_nodes <- go_processes |>
   120	  dplyr::transmute(
   121	    id = paste0("go::", go_term), label = go_term, group = "BiologicalProcess", shape = "dot", size = node_size,
   122	    title = paste0("<b>GO biological process:</b> ", go_term)
   123	  )
   124	compound_nodes <- compound_rank_data |>
   125	  dplyr::transmute(
   126	    id = paste0("compound::", molecule_chembl_id),
   127	    label = rank_label,
   128	    group = "Compound",
   129	    shape = "diamond",
   130	    size = compound_size,
   131	    borderWidth = compound_border_width,
   132	    shadow = Map(function(enabled, color, size) list(enabled = enabled, color = color, size = size, x = 0, y = 0), TRUE, compound_shadow_color, compound_shadow_size),
   133	    title = paste0(
   134	      "<div style='font-family:Calibri,Arial,sans-serif;line-height:1.4;'>",
   135	      "<strong>Compound:</strong> ", name, "<br>",
   136	      "<strong>ChEMBL ID:</strong> ", molecule_chembl_id, "<br><br>",
   137	      "<strong>Direct EEF1A1 rank:</strong> ", direct_rank, "<br>",
   138	      "<strong>Translation-network rank:</strong> ", network_rank, "<br>",
   139	      "<strong>Direct score:</strong> ", round(direct_total_score, 2), " / 16<br>",
   140	      "<strong>Network score:</strong> ", round(network_total_score, 2), "<br>",
   141	      "<strong>Priority category:</strong> ", priority_category, "<br>",
   142	      "<strong>Direct primary target:</strong> ", direct_primary_target, "<br>",
   143	      "<strong>Direct relationship:</strong> ", direct_primary_relationship, "<br>",
   144	      "<strong>Direct minimum activity:</strong> ", signif(direct_minimum_activity_nM, 4), " nM<br>",
   145	      "<strong>Strongest network target:</strong> ", strongest_network_target, "<br>",
   146	      "<strong>Network target count:</strong> ", network_target_count,
   147	      "</div>"
   148	    )
   149	  )
   150	
   151	nodes <- dplyr::bind_rows(protein_nodes, pathway_nodes, go_nodes, compound_nodes)
   152	
   153	ppi_edges <- ppi |>
   154	  dplyr::transmute(from = paste0("protein::", source), to = paste0("protein::", target), color = "#9A9A9A", width = 0.9 + 2.7 * score, arrows = "")
   155	pathway_edges <- protein_pathway |>
   156	  dplyr::transmute(from = paste0("protein::", protein), to = paste0("pathway::", pathway), color = "#40916C", width = 1.15, arrows = "")
   157	go_edges <- protein_go |>
   158	  dplyr::transmute(from = paste0("protein::", protein), to = paste0("go::", biological_process), color = "#5B8FD1", width = 1.05, arrows = "")
   159	compound_edges <- compound_edges_data |>
   160	  dplyr::mutate(
   161	    edge_color = dplyr::case_when(
   162	      relationship == "HAS_INHIBITORY_ACTIVITY_AGAINST" ~ "#C43C39",
   163	      relationship == "BINDS_TO" ~ "#8E63B0",
   164	      TRUE ~ "#C77729"
   165	    ),
   166	    edge_width = dplyr::case_when(
   167	      relationship == "HAS_INHIBITORY_ACTIVITY_AGAINST" ~ 3,
   168	      relationship == "BINDS_TO" ~ 2.4,
   169	      TRUE ~ 1.7
   170	    )
   171	  ) |>
   172	  dplyr::transmute(
   173	    from = paste0("compound::", molecule_chembl_id),
   174	    to = paste0("protein::", queried_protein),
   175	    color = edge_color, width = edge_width, arrows = "",
   176	    title = paste0("<b>", relationship, "</b><br>", display_name, " - ", queried_protein, "<br>Minimum activity: ", signif(minimum_activity_nM, 4), " nM")
   177	  )
   178	edges <- dplyr::bind_rows(ppi_edges, pathway_edges, go_edges, compound_edges)
   179	
   180	network <- visNetwork::visNetwork(nodes, edges, width = "100%", height = "900px", main = "Ranked eEF1A Compound Network") |>
   181	  visNetwork::visGroups(groupname = "Compound", shape = "diamond", color = list(background = "#E58B3A", border = "#713900"), font = list(face = "Calibri", size = 14)) |>
   182	  visNetwork::visGroups(groupname = "Protein", shape = "dot", color = list(background = "#D95F59", border = "#8E2F2A"), font = list(face = "Calibri", size = 16)) |>
   183	  visNetwork::visGroups(groupname = "Pathway", shape = "dot", color = list(background = "#4E9F6E", border = "#25633A"), font = list(face = "Calibri", size = 14)) |>
   184	  visNetwork::visGroups(groupname = "BiologicalProcess", shape = "dot", color = list(background = "#5B8FD1", border = "#2B5797"), font = list(face = "Calibri", size = 13)) |>
   185	  visNetwork::visNodes(font = list(face = "Calibri", strokeWidth = 3, strokeColor = "#FFFFFF")) |>
   186	  visNetwork::visEdges(arrows = "", smooth = FALSE, selectionWidth = 3, hoverWidth = 2) |>
   187	  visNetwork::visOptions(
   188	    highlightNearest = list(enabled = TRUE, degree = 1, hover = FALSE, algorithm = "all"),
   189	    nodesIdSelection = list(enabled = TRUE, useLabels = TRUE, main = "Find a node"),
   190	    selectedBy = list(variable = "group", main = "Filter by node type", multiple = TRUE)
   191	  ) |>
   192	  visNetwork::visInteraction(hover = TRUE, navigationButtons = TRUE, keyboard = TRUE, hideEdgesOnDrag = TRUE) |>
   193	  visNetwork::visPhysics(
   194	    solver = "forceAtlas2Based",
   195	    forceAtlas2Based = list(gravitationalConstant = -110, centralGravity = 0.006, springLength = 265, springConstant = 0.033, damping = 0.65, avoidOverlap = 0.8),
   196	    stabilization = list(enabled = TRUE, iterations = 3500, fit = TRUE)
   197	  ) |>
   198	  visNetwork::visLayout(randomSeed = 42, improvedLayout = TRUE) |>
   199	  visNetwork::visEvents(stabilizationIterationsDone = "function(){this.setOptions({physics:{enabled:false}});this.fit();}")

   201	# Rendering helpers for table panels.
   202	html_table <- function(data, columns, labels) {
   203	  shown <- data |> dplyr::select(dplyr::all_of(columns))
   204	  names(shown) <- labels
   205	  htmltools::tags$table(
   206	    class = "ranking-table",
   207	    htmltools::tags$thead(htmltools::tags$tr(lapply(names(shown), htmltools::tags$th))),
   208	    htmltools::tags$tbody(lapply(seq_len(nrow(shown)), function(i) {
   209	      htmltools::tags$tr(lapply(shown[i, , drop = FALSE], function(x) htmltools::tags$td(as.character(x))))
   210	    }))
   211	  )
   212	}

   220	    htmltools::tags$div(class = "symbol diamond orange thick"), htmltools::tags$div("Thick dark border"), htmltools::tags$div("Top-ranked direct candidate"),
   221	    htmltools::tags$div(class = "symbol diamond orange halo"), htmltools::tags$div("Dark halo"), htmltools::tags$div("Top-ranked translation-network candidate"),
   222	    htmltools::tags$div(class = "tooltip-symbol", "i"), htmltools::tags$div("Tooltip"), htmltools::tags$div("Full ranking evidence"),
   223	    htmltools::tags$div(class = "label-symbol", "D1 / N2"), htmltools::tags$div("Label prefix"), htmltools::tags$div("Direct and network rank"),
   224	    htmltools::tags$div(class = "opacity-symbol"), htmltools::tags$div("Reduced opacity"), htmltools::tags$div("Unrelated nodes after selection")
   225	  )
   226	)
   227	
   228	leaderboards <- htmltools::tags$div(
   229	  class = "ranking-section",
   230	  htmltools::tags$h2("Candidate rankings"),
   231	  htmltools::tags$div(class = "leaderboard-grid",
   232	    htmltools::tags$div(class = "leaderboard-card",
   233	      htmltools::tags$h3("Direct EEF1A1 ranking"),
   234	      html_table(
   235	        direct_ranking,

   237	        c("Rank", "Compound", "Score", "Primary target", "Evidence", "Minimum nM")
   238	      )

   240	    htmltools::tags$div(class = "leaderboard-card",
   241	      htmltools::tags$h3("Translation-network ranking"),
   242	      html_table(

   245	        c("Rank", "Compound", "Score", "Strongest target", "Targets", "Minimum nM")
   246	      )
   247	    )
   248	  ),
   249	  rank_visual_key,
   250	  htmltools::tags$p(class = "report-link", "Detailed score decomposition: results/A15_candidate_rankings.html")
   251	)

   253	page_style <- htmltools::tags$style(htmltools::HTML("\nbody{margin:0;padding:14px;background:#fff;font-family:Calibri,Arial,sans-serif;color:#222;}\nh1,h2,h3{font-weight:600;} select,input,button,label{font-family:Calibri,Arial,sans-serif!important;}\n.vis-network{border:1px solid #e0e0e0;border-radius:5px;background:#fff;}\n.ranking-section{margin-top:18px;}\n.leaderboard-grid{display:grid;grid-template-columns:repeat(auto-fit,minmax(520px,1fr));gap:18px;}\n.leaderboard-card,.rank-key{border:1px solid #d7dade;border-radius:6px;background:#f8f9fb;padding:14px;}\n.ranking-table{border-collapse:collapse;width:100%;font-size:12px;background:#fff;}\n.ranking-table th{background:#e9eef3;text-align:left;padding:7px;border:1px solid #ccd3da;}\n.ranking-table td{padding:7px;border:1px solid #d9dee3;}\n.rank-key{margin-top:18px;}\n.rank-key-grid{display:grid;grid-template-columns:70px 150px 1fr;gap:10px 12px;align-items:center;}\n.symbol{width:18px;height:18px;margin-left:14px;}\n.diamond{transform:rotate(45deg);border-radius:2px;} .orange{background:#E58B3A;border:2px solid #713900;}\n.large{width:28px;height:28px;margin-left:9px;} .thick{border-width:5px;}\n.halo{box-shadow:0 0 0 5px rgba(70,35,120,.45),0 0 14px 7px rgba(70,35,120,.4);}\n.tooltip-symbol{width:24px;height:24px;border-radius:50%;background:#eef2f6;border:1px solid #8996a3;text-align:center;line-height:24px;font-weight:bold;margin-left:10px;}\n.label-symbol{font-weight:600;color:#713900;} .opacity-symbol{width:30px;height:5px;background:rgba(154,154,154,.2);margin-left:6px;}\n.report-link{font-size:12px;color:#555;}\n"))

   255	network <- htmlwidgets::appendContent(network, leaderboards)
   256	network <- htmlwidgets::prependContent(network, page_style)
   257	htmlwidgets::saveWidget(network, output_file, selfcontained = TRUE)
   258	
   259	if (!file.exists(output_file) || file.info(output_file)$size <= 0) stop("Ranked network HTML was not created.")
   260	cat("\nA15B ranked network created successfully.\n")
   261	cat("Output: ", output_file, "\n", sep = "")
   262	utils::browseURL(normalizePath(output_file, winslash = "/", mustWork = TRUE))
